@OptIn(ExperimentalCoroutinesApi::class)
class ExampleUnitTest {

  private val testDispatcher = StandardTestDispatcher()
  private lateinit var fakeSqlite: InMemorySqliteStore
  private lateinit var fakePrefs: InMemorySharedPrefsStore
  private lateinit var repository: NotesRepository
  private var fakeTime = 1_000L

  @Before
  fun setUp() {
    Dispatchers.setMain(testDispatcher)
    fakeSqlite = InMemorySqliteStore()
    fakePrefs = InMemorySharedPrefsStore()
    repository =
      NotesRepository(
        sqliteStore = fakeSqlite,
        sharedPrefsStore = fakePrefs,
        ioDispatcher = testDispatcher,
        clock = { ++fakeTime },
      )
  }

  @After
  fun tearDown() {
    Dispatchers.resetMain()
  }

  @Test
  fun `crud and in-place edit preserve uuid and original storage engine`() = runTest {
    assertTrue(repository.createNote("  SQLite Note  ", "  Body A  ", StorageMethod.SQLITE))
    assertTrue(repository.createNote("Prefs Note", "Body B", StorageMethod.SHARED_PREFERENCES))

    val initial = repository.loadSnapshot()
    assertEquals(2, initial.allNotes.size)
    assertEquals(1, initial.sqliteCount)
    assertEquals(1, initial.sharedPrefsCount)

    val sqNote = initial.sqliteNotes.single()
    assertEquals("SQLite Note", sqNote.title)

  private class InMemorySqliteStore : SqliteNoteStore {
    private val notes = linkedMapOf<String, SqliteNoteRecord>()
    private val tombstones = linkedMapOf<String, Long>()
    var failNextFinalize = false

    override fun getAllNotes(): List<SqliteNoteRecord> =
      notes.values.sortedByDescending { it.updatedAt }

    override fun getPendingDeletedIds(): Set<String> = tombstones.keys.toSet()

    override fun insertNote(record: SqliteNoteRecord): Boolean {
      notes[record.id] = record
      tombstones.remove(record.id)
      return true
    }

    override fun updateNote(id: String, title: String, content: String, updatedAt: Long): Boolean {
      val existing = notes[id] ?: return false
      notes[id] = existing.copy(title = title, content = content, updatedAt = updatedAt)
      return true
    }

    override fun deleteNoteWithTombstone(
      noteId: String,
      hasSyncedCopyInPrefs: Boolean,
      deletedAt: Long,
    ): Boolean {
      val removed = notes.remove(noteId)
      if (removed == null && !hasSyncedCopyInPrefs) return false
      if (removed?.isSynced == true || hasSyncedCopyInPrefs) {
        tombstones[noteId] = deletedAt
      }
      return true
    }

    override fun finalizeSyncMetadata(
      syncedNoteIds: List<String>,
      processedTombstoneIds: Set<String>,
      syncTimestamp: Long,
    ) {
      if (failNextFinalize) {
        failNextFinalize = false
        throw IllegalStateException("Simulated SQLite error")
      }
      processedTombstoneIds.forEach { tombstones.remove(it) }
      syncedNoteIds.forEach { id ->
        notes[id]?.let { existing ->
          notes[id] = existing.copy(isSynced = true, lastSyncedAt = syncTimestamp)
        }
      }
    }
  }

  private class InMemorySharedPrefsStore : SharedPrefsNoteStore {
    private val store = linkedMapOf<String, SharedPrefsNoteRecord>()
    var failNextCommit = false

    override fun getAllNotes(): List<SharedPrefsNoteRecord> =
      store.values.sortedByDescending { it.updatedAt }

    override fun getNoteById(id: String): SharedPrefsNoteRecord? = store[id]

    override fun saveNote(record: SharedPrefsNoteRecord): Boolean {
      store[record.id] = record
      return true
    }

    override fun deleteNote(id: String): Boolean = store.remove(id) != null

    override fun applySyncBatch(
      removals: Set<String>,
      upserts: List<SharedPrefsNoteRecord>,
    ): Boolean {
      if (failNextCommit) {
        failNextCommit = false
        return false
      }
      removals.forEach { store.remove(it) }
      upserts.forEach { store[it.id] = it }
      return true
    }
  }
}

# FolioNotes — Dual-Storage Android Notes Application (Final Plan)

FolioNotes is a native Kotlin and Jetpack Compose note-taking application built for MADT Lab #5 that demonstrates complete local CRUD operations, user-selectable persistence across **SQLite (`SQLiteOpenHelper`)** and **`SharedPreferences`**, **two-stage recoverable SQLite-to-SharedPreferences synchronization** with persistent deletion tracking, strict input validation, and full Android string-resource localization.

## User Review & Critical Decisions

> [!IMPORTANT]
> **Confirmed Architectural Decisions & Final Scope**
> - **Primary SQLite Engine**: Native `SQLiteOpenHelper` for direct schema management, local SQLite transactions, and parameterized queries.
> - **Immutable Storage Location After Creation**: Users select **SQLite** or **SharedPreferences** when creating a new note. Editing an existing note preserves its original storage location and updates the existing record in place by UUID—no cross-storage migration is performed.
> - **Storage Visibility**: Filter tabs (**All**, **SQLite**, **SharedPreferences**) with origin/sync badges on every note card.
> - **Two-Stage Recoverable Synchronization**: Stage 1 commits `SharedPreferences` (`editor.commit() == true`), and only then does Stage 2 finalize SQLite sync metadata and clear processed `deleted_notes` tombstones.
> - **Strict Origin Protection & Collision Safety**: Sync updates `SYNCED_FROM_SQLITE` entries from SQLite, strictly preserves `SHARED_PREFERENCES` entries, and safely re-keys any unexpected UUID collision without overwriting independent `SharedPreferences` notes.
> - **Localization Scope**: English string resources (`values/strings.xml`) covering 100% of UI text, validation errors, dialogs, snackbars, and accessibility descriptions.

---

## 1. Overview & Core Concept

- **What It Does**: Lets users create, read, inspect, edit, and delete notes composed of a title and content. Users select **SQLite** or **SharedPreferences** upon note creation, filter notes by storage source, and run an explicit **SQLite → SharedPreferences** sync with recoverable failure handling and origin protection.
- **Target Audience / Persona**: Mobile Application Development students and instructors evaluating MADT Lab #5 (Activities, Interactions, and Storage).
- **Key Value**: Keeps storage boundaries clean and predictable while thoroughly handling persistence across restarts, empty-field validation, full string-resource localization, and sync edge cases.

### Official MADT Lab #5 Requirements Verification *(7 Special Part + 7 Common Part)*

#### Special Part — 7 Core Requirements

| # | Special Part Requirement | Planned Implementation |

## 3. Key Product Decisions & Trade-Offs

- **Decision 1 — Native Kotlin, Jetpack Compose, Local-Only Persistence & Localization**:
  - *Chosen Approach*: Pure native Android using Kotlin, Jetpack Compose (Material 3), and an MVVM + Repository architecture backed strictly by local `SQLiteOpenHelper` and `SharedPreferences` (zero Firebase or cloud storage). Every visible UI string, error, badge, dialog, snackbar, and accessibility description is externalized into `res/values/strings.xml`.
  - *Why*: Ensures durable persistence across app restarts, clean maintainability for subsequent code review, and full compliance with local storage and string-resource localization requirements.

- **Decision 2 — Preserve Original Storage Location on Edit (No Cross-Storage Migration)**:
  - *Chosen Approach*: Users choose **SQLite** or **SharedPreferences** during note creation. When editing an existing note, the storage location is read-only and updates are written in place to that same store under the note's existing UUID.
  - *Why*: Eliminates partial-failure and duplicate-record risks from cross-engine moves, avoids unnecessary migration infrastructure, and keeps **SQLite → SharedPreferences** synchronization as a distinct, explicit feature.

- **Decision 3 — Two-Stage Recoverable Sync (No Cross-Storage Atomic Illusion)**:
  - *Chosen Approach*: Treat SQLite and `SharedPreferences` as separate local engines synchronized via a two-stage recoverable sequence under a coroutine `Mutex`: Stage 1 commits `SharedPreferences` (`editor.commit() == true`), and Stage 2 finalizes SQLite metadata (`is_synced`, `deleted_notes`) inside a local SQLite transaction.
  - *Why*: SQLite and `SharedPreferences` cannot participate in one shared atomic transaction. Ordering `SharedPreferences` commit first guarantees that if Stage 1 fails, SQLite retains all pending flags and tombstones; if Stage 2 fails after Stage 1 succeeds, pending metadata remains in SQLite so retrying sync is safe and idempotent.

- **Decision 4 — Origin-Protected `SharedPreferences` & Unexpected UUID Collision Handling**:
  - *Chosen Approach*: Tag every `SharedPreferences` JSON payload with `"origin": "SHARED_PREFERENCES" | "SYNCED_FROM_SQLITE"`. Sync only updates or deletes entries with `origin == SYNCED_FROM_SQLITE`. If an active SQLite note's UUID ever unexpectedly collides with an existing `SHARED_PREFERENCES`-origin entry, sync never overwrites the independent SP note—it re-keys the `SHARED_PREFERENCES` note to a fresh unused UUID in the same batch before writing the `SYNCED_FROM_SQLITE` entry.
  - *Why*: Guarantees independently created `SharedPreferences` notes are never silently overwritten or lost.

- **Decision 5 — Persistent Deletion Tombstones (`deleted_notes`)**:
  - *Chosen Approach*: When deleting a previously synchronized SQLite note (`is_synced == 1` or having a `SYNCED_FROM_SQLITE` copy), atomically delete from `notes` and insert the UUID into `deleted_notes` within one SQLite transaction.
  - *Why*: Survives process death and app restarts until Stage 1 + Stage 2 synchronization completes.

### Database Schema & Data Model

#### SQLite Schema (`NotesDbHelper`, `DATABASE_VERSION = 1`)

1. **`notes` Table**:
   - `id TEXT PRIMARY KEY` (UUID string)
   - `title TEXT NOT NULL`
   - `content TEXT NOT NULL`
   - `created_at INTEGER NOT NULL`
   - `updated_at INTEGER NOT NULL`
   - `is_synced INTEGER NOT NULL DEFAULT 0` (`1` only after Stage 1 commit + Stage 2 SQLite finalization)
   - `last_synced_at INTEGER` (nullable epoch millis)

2. **`deleted_notes` Table (Persistent Deletion Tombstones)**:
   - `note_id TEXT PRIMARY KEY` (UUID of deleted, previously synchronized SQLite note)
   - `deleted_at INTEGER NOT NULL` (epoch millis)

#### `SharedPreferences` JSON Schema (`notes_prefs`)

- **Key**: `note.id` (UUID string)
- **Value**: JSON string with fields:
  - `id`: String (UUID)
  - `title`: String
  - `content`: String
  - `createdAt`: Long
  - `updatedAt`: Long
  - `origin`: `"SHARED_PREFERENCES"` or `"SYNCED_FROM_SQLITE"`

### Two-Stage Recoverable Synchronization Flow

All repository mutations and sync operations run inside `Mutex.withLock` on `Dispatchers.IO`:

1. **Snapshot Read**:
   - Read all active rows from SQLite `notes`, all pending UUIDs from `deleted_notes`, and all valid JSON records from `SharedPreferences`.
2. **Stage 1 — `SharedPreferences` Batch Build & Verified Commit**:
   - Open `prefs.edit()`:
     - **Tombstone Deletions**: For each `tombstoneId` in `deleted_notes` (excluding any ID present in active `notes`):
       - If `SharedPreferences` contains `tombstoneId` with `origin == SYNCED_FROM_SQLITE`, call `editor.remove(tombstoneId)`.
       - If `SharedPreferences` contains `tombstoneId` with `origin == SHARED_PREFERENCES`, **do NOT remove it** (preserve SP-only note).
     - **SQLite Authoritative Upserts**: For each active `sqliteNote` in `notes`:
       - Check existing `SharedPreferences` entry at `sqliteNote.id`:
         - *Absent*: Put JSON with `origin = SYNCED_FROM_SQLITE` (count as inserted).
         - *Present & `origin == SYNCED_FROM_SQLITE`*: Overwrite with latest SQLite fields and `origin = SYNCED_FROM_SQLITE` (count as updated only if fields changed; repeated runs are no-ops).
         - *Present & `origin == SHARED_PREFERENCES` (Unexpected UUID Collision)*: **Never overwrite.** Generate a fresh unused UUID `rekeyId`, write the untouched `SHARED_PREFERENCES` note under `rekeyId` (with `id = rekeyId` and `origin = SHARED_PREFERENCES`), and write `sqliteNote` under `sqliteNote.id` with `origin = SYNCED_FROM_SQLITE`.
   - Execute `val committed = editor.commit()`.
   - **If `!committed` (or exception)**: Return `SyncResult.Failure`. Stage 2 is skipped; SQLite `is_synced` and `deleted_notes` remain unchanged for safe retry.
3. **Stage 2 — SQLite Metadata Finalization**:
   - Inside a local SQLite transaction (`beginTransaction()`):
     - Delete all processed `tombstoneId` rows from `deleted_notes`.
     - Update active synced rows in `notes` with `is_synced = 1` and `last_synced_at = syncTimestamp`.
     - Mark transaction successful.
   - **If Stage 2 fails**: Catch the error and return `SyncResult.Failure`. Pending `deleted_notes` and `is_synced = 0` flags remain in SQLite. On the next sync retry, Stage 1 idempotently produces the exact same `SharedPreferences` state (zero duplicates, zero lost SP-only notes), and Stage 2 completes SQLite finalization cleanly.

### Data Integrity Guarantees

- **In-Place Editing**: Preserves the note's UUID and its original storage engine (`SQLite` or `SharedPreferences`), updating only the existing row or key without creating duplicates or moving records across engines.
- **Persistent Deletion Tracking**: Deleting a previously synchronized SQLite note atomically removes it from `notes` and records its UUID in `deleted_notes` within one SQLite transaction, surviving restarts until Stage 1 + Stage 2 sync completes.
- **Idempotent Repeated Synchronization**: Keying both SQLite and `SharedPreferences` by UUID guarantees repeated syncs converge to the exact same state with zero duplicates.
- **Safe Storage Access**: Parameterized SQL (`?` bindings), `.use {}` cursor scoping, defensive JSON parsing, and `Mutex` + `Dispatchers.IO` serialization.

package com.example.data.local

import android.content.ContentValues
import android.database.sqlite.SQLiteDatabase
import com.example.data.local.NotesDbHelper.Companion.COL_CONTENT
import com.example.data.local.NotesDbHelper.Companion.COL_CREATED_AT
import com.example.data.local.NotesDbHelper.Companion.COL_DELETED_AT
import com.example.data.local.NotesDbHelper.Companion.COL_DELETED_NOTE_ID
import com.example.data.local.NotesDbHelper.Companion.COL_ID
import com.example.data.local.NotesDbHelper.Companion.COL_IS_SYNCED
import com.example.data.local.NotesDbHelper.Companion.COL_LAST_SYNCED_AT
import com.example.data.local.NotesDbHelper.Companion.COL_TITLE
import com.example.data.local.NotesDbHelper.Companion.COL_UPDATED_AT
import com.example.data.local.NotesDbHelper.Companion.TABLE_DELETED_NOTES
import com.example.data.local.NotesDbHelper.Companion.TABLE_NOTES
import com.example.data.model.SqliteNoteRecord

interface SqliteNoteStore {
  fun getAllNotes(): List<SqliteNoteRecord>

  fun getPendingDeletedIds(): Set<String>

  fun insertNote(record: SqliteNoteRecord): Boolean

  fun updateNote(id: String, title: String, content: String, updatedAt: Long): Boolean

  fun deleteNoteWithTombstone(
    noteId: String,
    hasSyncedCopyInPrefs: Boolean,
    deletedAt: Long,
  ): Boolean

  fun finalizeSyncMetadata(
    syncedNoteIds: List<String>,
    processedTombstoneIds: Set<String>,
    syncTimestamp: Long,
  )
}

class SQLiteDataSource(private val dbHelper: NotesDbHelper) : SqliteNoteStore {

  override fun getAllNotes(): List<SqliteNoteRecord> {
    val db = dbHelper.readableDatabase
    val results = mutableListOf<SqliteNoteRecord>()
    val columns =
      arrayOf(
        COL_ID,
        COL_TITLE,
        COL_CONTENT,
        COL_CREATED_AT,
        COL_UPDATED_AT,
        COL_IS_SYNCED,
        COL_LAST_SYNCED_AT,
      )
    db.query(TABLE_NOTES, columns, null, null, null, null, "$COL_UPDATED_AT DESC").use { cursor ->
      val idIdx = cursor.getColumnIndexOrThrow(COL_ID)
      val titleIdx = cursor.getColumnIndexOrThrow(COL_TITLE)
      val contentIdx = cursor.getColumnIndexOrThrow(COL_CONTENT)
      val createdIdx = cursor.getColumnIndexOrThrow(COL_CREATED_AT)
      val updatedIdx = cursor.getColumnIndexOrThrow(COL_UPDATED_AT)
      val syncedIdx = cursor.getColumnIndexOrThrow(COL_IS_SYNCED)
      val lastSyncedIdx = cursor.getColumnIndexOrThrow(COL_LAST_SYNCED_AT)

      while (cursor.moveToNext()) {
        results.add(
          SqliteNoteRecord(
            id = cursor.getString(idIdx),
            title = cursor.getString(titleIdx),
            content = cursor.getString(contentIdx),
            createdAt = cursor.getLong(createdIdx),
            updatedAt = cursor.getLong(updatedIdx),
            isSynced = cursor.getInt(syncedIdx) == 1,
            lastSyncedAt = if (cursor.isNull(lastSyncedIdx)) null else cursor.getLong(lastSyncedIdx),
          )
        )
      }
    }
    return results
  }

  override fun getPendingDeletedIds(): Set<String> {
    val db = dbHelper.readableDatabase
    val ids = linkedSetOf<String>()
    db.query(
        TABLE_DELETED_NOTES,
        arrayOf(COL_DELETED_NOTE_ID),
        null,
        null,
        null,
        null,
        null,
      )
      .use { cursor ->
        val idIdx = cursor.getColumnIndexOrThrow(COL_DELETED_NOTE_ID)
        while (cursor.moveToNext()) {
          ids.add(cursor.getString(idIdx))
        }
      }
    return ids
  }

  override fun insertNote(record: SqliteNoteRecord): Boolean {
    val db = dbHelper.writableDatabase
    db.beginTransaction()
    return try {
      val values =
        ContentValues().apply {
          put(COL_ID, record.id)
          put(COL_TITLE, record.title)
          put(COL_CONTENT, record.content)
          put(COL_CREATED_AT, record.createdAt)
          put(COL_UPDATED_AT, record.updatedAt)
          put(COL_IS_SYNCED, if (record.isSynced) 1 else 0)
          if (record.lastSyncedAt != null) {
            put(COL_LAST_SYNCED_AT, record.lastSyncedAt)
          } else {
            putNull(COL_LAST_SYNCED_AT)
          }
        }
      val rowId = db.insertWithOnConflict(TABLE_NOTES, null, values, SQLiteDatabase.CONFLICT_ABORT)
      if (rowId == -1L) return false
      db.delete(TABLE_DELETED_NOTES, "$COL_DELETED_NOTE_ID = ?", arrayOf(record.id))
      db.setTransactionSuccessful()
      true
    } finally {
      db.endTransaction()
    }
  }

  override fun updateNote(id: String, title: String, content: String, updatedAt: Long): Boolean {
    val db = dbHelper.writableDatabase
    val values =
      ContentValues().apply {
        put(COL_TITLE, title)
        put(COL_CONTENT, content)
        put(COL_UPDATED_AT, updatedAt)
      }
    val updatedRows = db.update(TABLE_NOTES, values, "$COL_ID = ?", arrayOf(id))
    return updatedRows > 0
  }

  override fun deleteNoteWithTombstone(
    noteId: String,
    hasSyncedCopyInPrefs: Boolean,
    deletedAt: Long,
  ): Boolean {
    val db = dbHelper.writableDatabase
    db.beginTransaction()
    return try {
      var wasPreviouslySynced = false
      db.query(
          TABLE_NOTES,
          arrayOf(COL_IS_SYNCED),
          "$COL_ID = ?",
          arrayOf(noteId),
          null,
          null,
          null,
        )
        .use { cursor ->
          if (cursor.moveToFirst()) {
            wasPreviouslySynced = cursor.getInt(cursor.getColumnIndexOrThrow(COL_IS_SYNCED)) == 1
          }
        }

      val deletedRows = db.delete(TABLE_NOTES, "$COL_ID = ?", arrayOf(noteId))
      if (deletedRows == 0 && !hasSyncedCopyInPrefs) {
        return false
      }

      if (wasPreviouslySynced || hasSyncedCopyInPrefs) {
        val tombstone =
          ContentValues().apply {
            put(COL_DELETED_NOTE_ID, noteId)
            put(COL_DELETED_AT, deletedAt)
          }
        db.insertWithOnConflict(
          TABLE_DELETED_NOTES,
          null,
          tombstone,
          SQLiteDatabase.CONFLICT_REPLACE,
        )
      }

      db.setTransactionSuccessful()
      true
    } finally {
      db.endTransaction()
    }
  }

  override fun finalizeSyncMetadata(
    syncedNoteIds: List<String>,
    processedTombstoneIds: Set<String>,
    syncTimestamp: Long,
  ) {
    if (syncedNoteIds.isEmpty() && processedTombstoneIds.isEmpty()) return

    val db = dbHelper.writableDatabase
    db.beginTransaction()
    try {
      for (tombstoneId in processedTombstoneIds) {
        db.delete(TABLE_DELETED_NOTES, "$COL_DELETED_NOTE_ID = ?", arrayOf(tombstoneId))
      }
      for (noteId in syncedNoteIds) {
        val values =
          ContentValues().apply {
            put(COL_IS_SYNCED, 1)
            put(COL_LAST_SYNCED_AT, syncTimestamp)
          }
        db.update(TABLE_NOTES, values, "$COL_ID = ?", arrayOf(noteId))
      }
      db.setTransactionSuccessful()
    } finally {
      db.endTransaction()
    }
  }
}

package com.example.data.local

import android.content.Context
import android.content.SharedPreferences
import com.example.data.model.PrefsOrigin
import com.example.data.model.SharedPrefsNoteRecord
import org.json.JSONException
import org.json.JSONObject

interface SharedPrefsNoteStore {
  fun getAllNotes(): List<SharedPrefsNoteRecord>

  fun getNoteById(id: String): SharedPrefsNoteRecord?

  fun saveNote(record: SharedPrefsNoteRecord): Boolean

  fun deleteNote(id: String): Boolean

  fun applySyncBatch(removals: Set<String>, upserts: List<SharedPrefsNoteRecord>): Boolean
}

class SharedPrefsDataSource(context: Context) : SharedPrefsNoteStore {

  private val prefs: SharedPreferences =
    context.applicationContext.getSharedPreferences(PREFS_FILE_NAME, Context.MODE_PRIVATE)

  override fun getAllNotes(): List<SharedPrefsNoteRecord> {
    val entries = prefs.all
    val notes = mutableListOf<SharedPrefsNoteRecord>()
    for ((key, value) in entries) {
      val rawJson = value as? String ?: continue
      val parsed = parseRecord(key, rawJson)
      if (parsed != null) {
        notes.add(parsed)
      }
    }
    return notes.sortedByDescending { it.updatedAt }
  }

  override fun getNoteById(id: String): SharedPrefsNoteRecord? {
    val rawJson = prefs.getString(id, null) ?: return null
    return parseRecord(id, rawJson)
  }

  override fun saveNote(record: SharedPrefsNoteRecord): Boolean {
    return prefs.edit().putString(record.id, encodeRecord(record)).commit()
  }

  override fun deleteNote(id: String): Boolean {
    return prefs.edit().remove(id).commit()
  }

  override fun applySyncBatch(
    removals: Set<String>,
    upserts: List<SharedPrefsNoteRecord>,
  ): Boolean {
    val editor = prefs.edit()
    for (id in removals) {
      editor.remove(id)
    }
    for (record in upserts) {
      editor.putString(record.id, encodeRecord(record))
    }
    return editor.commit()
  }

  private fun encodeRecord(record: SharedPrefsNoteRecord): String {
    return JSONObject()
      .apply {
        put(KEY_ID, record.id)
        put(KEY_TITLE, record.title)
        put(KEY_CONTENT, record.content)
        put(KEY_CREATED_AT, record.createdAt)
        put(KEY_UPDATED_AT, record.updatedAt)
        put(KEY_ORIGIN, record.origin.name)
      }
      .toString()
  }

  private fun parseRecord(fallbackId: String, rawJson: String): SharedPrefsNoteRecord? {
    return try {
      val obj = JSONObject(rawJson)
      val id = obj.optString(KEY_ID, fallbackId).ifBlank { fallbackId }
      val title = obj.getString(KEY_TITLE)
      val content = obj.getString(KEY_CONTENT)
      val createdAt = obj.getLong(KEY_CREATED_AT)
      val updatedAt = obj.getLong(KEY_UPDATED_AT)
      val originRaw = obj.optString(KEY_ORIGIN, PrefsOrigin.SHARED_PREFERENCES.name)
      val origin =
        PrefsOrigin.entries.firstOrNull { it.name == originRaw } ?: PrefsOrigin.SHARED_PREFERENCES
      SharedPrefsNoteRecord(
        id = id,
        title = title,
        content = content,
        createdAt = createdAt,
        updatedAt = updatedAt,
        origin = origin,
      )
    } catch (_: JSONException) {
      null
    }
  }

  companion object {
    const val PREFS_FILE_NAME = "notes_prefs"
    private const val KEY_ID = "id"
    private const val KEY_TITLE = "title"
    private const val KEY_CONTENT = "content"
    private const val KEY_CREATED_AT = "createdAt"
    private const val KEY_UPDATED_AT = "updatedAt"
    private const val KEY_ORIGIN = "origin"
  }
}

package com.example.data.repository

import com.example.data.local.SharedPrefsNoteStore
import com.example.data.local.SqliteNoteStore
import com.example.data.model.Note
import com.example.data.model.NoteDisplaySource
import com.example.data.model.NotesSnapshot
import com.example.data.model.PrefsOrigin
import com.example.data.model.SharedPrefsNoteRecord
import com.example.data.model.SqliteNoteRecord
import com.example.data.model.StorageMethod
import com.example.data.model.SyncResult
import java.util.UUID
import kotlinx.coroutines.CoroutineDispatcher
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.sync.Mutex
import kotlinx.coroutines.sync.withLock
import kotlinx.coroutines.withContext

class NotesRepository(
  private val sqliteStore: SqliteNoteStore,
  private val sharedPrefsStore: SharedPrefsNoteStore,
  private val ioDispatcher: CoroutineDispatcher = Dispatchers.IO,
  private val clock: () -> Long = { System.currentTimeMillis() },
  private val idGenerator: () -> String = { UUID.randomUUID().toString() },
) {
  private val mutex = Mutex()

  suspend fun loadSnapshot(): NotesSnapshot =
    withContext(ioDispatcher) {
      mutex.withLock { readSnapshotLocked() }
    }

  suspend fun createNote(
    title: String,
    content: String,
    storageMethod: StorageMethod,
  ): Boolean =
    withContext(ioDispatcher) {
      mutex.withLock {
        val now = clock()
        val id = generateUniqueIdLocked()
        val trimmedTitle = title.trim()
        val trimmedContent = content.trim()
        when (storageMethod) {
          StorageMethod.SQLITE -> {
            sqliteStore.insertNote(
              SqliteNoteRecord(
                id = id,
                title = trimmedTitle,
                content = trimmedContent,
                createdAt = now,
                updatedAt = now,
                isSynced = false,
                lastSyncedAt = null,
              )
            )
          }
          StorageMethod.SHARED_PREFERENCES -> {
            sharedPrefsStore.saveNote(
              SharedPrefsNoteRecord(
                id = id,
                title = trimmedTitle,
                content = trimmedContent,
                createdAt = now,
                updatedAt = now,
                origin = PrefsOrigin.SHARED_PREFERENCES,
              )
            )
          }
        }
      }
    }

  suspend fun updateNote(
    id: String,
    title: String,
    content: String,
    storageMethod: StorageMethod,
  ): Boolean =
    withContext(ioDispatcher) {
      mutex.withLock {
        val now = clock()
        val trimmedTitle = title.trim()
        val trimmedContent = content.trim()
        when (storageMethod) {
          StorageMethod.SQLITE -> {
            sqliteStore.updateNote(
              id = id,
              title = trimmedTitle,
              content = trimmedContent,
              updatedAt = now,
            )
          }
          StorageMethod.SHARED_PREFERENCES -> {
            val existing = sharedPrefsStore.getNoteById(id) ?: return@withLock false
            sharedPrefsStore.saveNote(
              existing.copy(
                title = trimmedTitle,
                content = trimmedContent,
                updatedAt = now,
                origin = PrefsOrigin.SHARED_PREFERENCES,
              )
            )
          }
        }
      }
    }

  suspend fun deleteNote(note: Note): Boolean =
    withContext(ioDispatcher) {
      mutex.withLock {
        when (note.storageMethod) {
          StorageMethod.SQLITE -> {
            val prefsEntry = sharedPrefsStore.getNoteById(note.id)
            val hasSyncedCopy = prefsEntry?.origin == PrefsOrigin.SYNCED_FROM_SQLITE
            sqliteStore.deleteNoteWithTombstone(
              noteId = note.id,
              hasSyncedCopyInPrefs = hasSyncedCopy,
              deletedAt = clock(),
            )
          }
          StorageMethod.SHARED_PREFERENCES -> {
            sharedPrefsStore.deleteNote(note.id)
          }
        }
      }
    }

  suspend fun syncSqliteToSharedPreferences(): SyncResult =
    withContext(ioDispatcher) {
      mutex.withLock {
        val sqliteNotes = sqliteStore.getAllNotes()
        val pendingDeletedIds = sqliteStore.getPendingDeletedIds()
        val prefsRecords = sharedPrefsStore.getAllNotes()

        val activeSqliteIds = sqliteNotes.map { it.id }.toSet()
        val prefsMap = prefsRecords.associateBy { it.id }
        val usedIds = (activeSqliteIds + prefsMap.keys).toMutableSet()

        val removals = mutableSetOf<String>()
        val upserts = mutableListOf<SharedPrefsNoteRecord>()
        val syncedNoteIds = sqliteNotes.map { it.id }

        var inserted = 0
        var updated = 0
        var deleted = 0
        var collisionRekeyed = 0

        // 1. Process deletion tombstones for notes removed from SQLite.
        for (tombstoneId in pendingDeletedIds) {
          if (tombstoneId in activeSqliteIds) continue
          val existingPrefs = prefsMap[tombstoneId]
          if (existingPrefs != null && existingPrefs.origin == PrefsOrigin.SYNCED_FROM_SQLITE) {
            removals.add(tombstoneId)
            deleted++
          }
        }

        // 2. Apply SQLite source-of-truth insertions and updates with origin protection.
        for (sq in sqliteNotes) {
          val desired =
            SharedPrefsNoteRecord(
              id = sq.id,
              title = sq.title,
              content = sq.content,
              createdAt = sq.createdAt,
              updatedAt = sq.updatedAt,
              origin = PrefsOrigin.SYNCED_FROM_SQLITE,
            )
          val existing = prefsMap[sq.id]
          when {
            existing == null -> {
              upserts.add(desired)
              inserted++
            }
            existing.origin == PrefsOrigin.SYNCED_FROM_SQLITE -> {
              if (existing != desired) {
                upserts.add(desired)
                updated++
              }
            }
            existing.origin == PrefsOrigin.SHARED_PREFERENCES -> {
              // Unexpected UUID collision: preserve independent SharedPreferences note under a new UUID.
              var rekeyId = idGenerator()
              while (!usedIds.add(rekeyId)) {
                rekeyId = idGenerator()
              }
              upserts.add(existing.copy(id = rekeyId, origin = PrefsOrigin.SHARED_PREFERENCES))
              upserts.add(desired)
              collisionRekeyed++
              inserted++
            }
          }
        }

        // Stage 1: Commit SharedPreferences batch synchronously and verify disk flush.
        if (removals.isNotEmpty() || upserts.isNotEmpty()) {
          val committed =
            try {
              sharedPrefsStore.applySyncBatch(removals = removals, upserts = upserts)
            } catch (_: Exception) {
              false
            }
          if (!committed) {
            return@withLock SyncResult.Failure
          }
        }

        // Stage 2: Finalize SQLite sync flags and clear processed tombstones only after Stage 1 succeeds.
        val needsSqliteFinalization =
          pendingDeletedIds.isNotEmpty() || sqliteNotes.any { !it.isSynced } || upserts.isNotEmpty()
        if (needsSqliteFinalization) {
          try {
            sqliteStore.finalizeSyncMetadata(
              syncedNoteIds = syncedNoteIds,
              processedTombstoneIds = pendingDeletedIds,
              syncTimestamp = clock(),
            )
          } catch (_: Exception) {
            // Pending tombstones and is_synced flags remain in SQLite for idempotent retry.
            return@withLock SyncResult.Failure
          }
        }

        SyncResult.Success(
          inserted = inserted,
          updated = updated,
          deleted = deleted,
          collisionRekeyed = collisionRekeyed,
        )
      }
    }

  private fun generateUniqueIdLocked(): String {
    val existingIds =
      (sqliteStore.getAllNotes().map { it.id } + sharedPrefsStore.getAllNotes().map { it.id })
        .toSet()
    var candidate = idGenerator()
    while (candidate in existingIds) {
      candidate = idGenerator()
    }
    return candidate
  }

  private fun readSnapshotLocked(): NotesSnapshot {
    val sqliteRecords = sqliteStore.getAllNotes()
    val pendingDeletedIds = sqliteStore.getPendingDeletedIds()
    val prefsRecords = sharedPrefsStore.getAllNotes()
    val prefsById = prefsRecords.associateBy { it.id }

    var pendingNoteSyncs = 0
    val sqliteUiNotes =
      sqliteRecords.map { sq ->
        val mirrored = prefsById[sq.id]?.takeIf { it.origin == PrefsOrigin.SYNCED_FROM_SQLITE }
        val isMirroredAndCurrent =
          sq.isSynced &&
            mirrored != null &&
            mirrored.title == sq.title &&
            mirrored.content == sq.content &&
            mirrored.updatedAt == sq.updatedAt &&
            mirrored.createdAt == sq.createdAt
        if (!isMirroredAndCurrent) {
          pendingNoteSyncs++
        }
        Note(
          id = sq.id,
          title = sq.title,
          content = sq.content,
          createdAt = sq.createdAt,
          updatedAt = sq.updatedAt,
          storageMethod = StorageMethod.SQLITE,
          isSynced = isMirroredAndCurrent,
          displaySource =
            if (mirrored != null) NoteDisplaySource.SYNCED_IN_BOTH else NoteDisplaySource.SQLITE_ONLY,
        )
      }

    val spOnlyUiNotes =
      prefsRecords
        .filter { it.origin == PrefsOrigin.SHARED_PREFERENCES }
        .map { sp ->
          Note(
            id = sp.id,
            title = sp.title,
            content = sp.content,
            createdAt = sp.createdAt,
            updatedAt = sp.updatedAt,
            storageMethod = StorageMethod.SHARED_PREFERENCES,
            isSynced = false,
            displaySource = NoteDisplaySource.SHARED_PREFS_ONLY,
          )
        }

    val sqliteMapById = sqliteUiNotes.associateBy { it.id }
    val prefsTabNotes =
      prefsRecords
        .map { sp ->
          when (sp.origin) {
            PrefsOrigin.SHARED_PREFERENCES ->
              Note(
                id = sp.id,
                title = sp.title,
                content = sp.content,
                createdAt = sp.createdAt,
                updatedAt = sp.updatedAt,
                storageMethod = StorageMethod.SHARED_PREFERENCES,
                isSynced = false,
                displaySource = NoteDisplaySource.SHARED_PREFS_ONLY,
              )
            PrefsOrigin.SYNCED_FROM_SQLITE -> {
              // SQLite remains the authoritative source of truth when a synced note exists in both.
              val sqliteOwner = sqliteMapById[sp.id]
              Note(
                id = sp.id,
                title = sqliteOwner?.title ?: sp.title,
                content = sqliteOwner?.content ?: sp.content,
                createdAt = sqliteOwner?.createdAt ?: sp.createdAt,
                updatedAt = sqliteOwner?.updatedAt ?: sp.updatedAt,
                storageMethod =
                  if (sqliteOwner != null) StorageMethod.SQLITE
                  else StorageMethod.SHARED_PREFERENCES,
                isSynced = sqliteOwner != null,
                displaySource =
                  if (sqliteOwner != null) NoteDisplaySource.SYNCED_IN_BOTH
                  else NoteDisplaySource.SHARED_PREFS_ONLY,
              )
            }
          }
        }
        .sortedByDescending { it.updatedAt }

    val allNotes = (sqliteUiNotes + spOnlyUiNotes).sortedByDescending { it.updatedAt }

    return NotesSnapshot(
      allNotes = allNotes,
      sqliteNotes = sqliteUiNotes,
      sharedPrefsNotes = prefsTabNotes,
      sqliteCount = sqliteUiNotes.size,
      sharedPrefsCount = prefsTabNotes.size,
      pendingSyncCount = pendingNoteSyncs + pendingDeletedIds.size,
    )
  }
}

class NotesViewModel(private val repository: NotesRepository) : ViewModel() {

  private val _uiState = MutableStateFlow(NotesUiState())
  val uiState: StateFlow<NotesUiState> = _uiState.asStateFlow()

  private val _feedbackEvents = Channel<UiFeedbackEvent>(Channel.BUFFERED)

    val titleError = if (editor.title.isBlank()) R.string.error_title_empty else null

        it.copy(
          editorState =
            editor.copy(
              titleErrorRes = titleError,
              contentErrorRes = contentError,
            )
        )
      }
      return
    }

    _uiState.update { it.copy(editorState = editor.copy(isSaving = true)) }

    viewModelScope.launch {
      val editId = editor.editingNoteId
      val success =
        if (editId == null) {
          repository.createNote(
            title = editor.title,
            content = editor.content,
            storageMethod = editor.storageMethod,
          )
        } else {
          repository.updateNote(
            id = editId,
            title = editor.title,
            content = editor.content,
            storageMethod = editor.storageMethod,
          )
        }

      val snapshot = repository.loadSnapshot()
      if (success) {
        _uiState.update {
          it.copy(
            isLoading = false,
            allNotes = snapshot.allNotes,
            sqliteNotes = snapshot.sqliteNotes,
            sharedPrefsNotes = snapshot.sharedPrefsNotes,
            sqliteCount = snapshot.sqliteCount,
            sharedPrefsCount = snapshot.sharedPrefsCount,
            pendingSyncCount = snapshot.pendingSyncCount,
            editorState = NoteEditorState(),
          )
        }
        val msgRes =
          when {
            editId != null -> R.string.msg_note_updated
            editor.storageMethod == StorageMethod.SQLITE -> R.string.msg_note_created_sqlite
            else -> R.string.msg_note_created_prefs
          }
        _feedbackEvents.send(UiFeedbackEvent(msgRes))
      } else {
        _uiState.update {
          it.copy(
            isLoading = false,
            allNotes = snapshot.allNotes,
            sqliteNotes = snapshot.sqliteNotes,
            sharedPrefsNotes = snapshot.sharedPrefsNotes,
            sqliteCount = snapshot.sqliteCount,
            sharedPrefsCount = snapshot.sharedPrefsCount,
            pendingSyncCount = snapshot.pendingSyncCount,
            editorState = it.editorState.copy(isSaving = false),
          )
        }
        _feedbackEvents.send(UiFeedbackEvent(R.string.msg_operation_failed))
      }
    }
  }

  fun requestDeleteNote(note: Note) {

  }

      val snapshot = repository.loadSnapshot()

      _feedbackEvents.send(
        UiFeedbackEvent(if (deleted) R.string.msg_note_deleted else R.string.msg_operation_failed)
      )
    }
  }

  fun syncNotes() {
    if (_uiState.value.isSyncing) return
    _uiState.update { it.copy(isSyncing = true) }

    viewModelScope.launch {
      val result = repository.syncSqliteToSharedPreferences()
      val snapshot = repository.loadSnapshot()
      _uiState.update {
        it.copy(
          isLoading = false,
          isSyncing = false,
          allNotes = snapshot.allNotes,
          sqliteNotes = snapshot.sqliteNotes,
          sharedPrefsNotes = snapshot.sharedPrefsNotes,
          sqliteCount = snapshot.sqliteCount,
          sharedPrefsCount = snapshot.sharedPrefsCount,
          pendingSyncCount = snapshot.pendingSyncCount,
        )
      }
      when (result) {
        is SyncResult.Success -> {
          if (result.totalChanges == 0) {
            _feedbackEvents.send(UiFeedbackEvent(R.string.msg_sync_already_up_to_date))
          } else {
            _feedbackEvents.send(
              UiFeedbackEvent(
                messageRes = R.string.msg_sync_success_format,
                formatArgs = listOf(result.inserted, result.updated, result.deleted),
              )
            )
          }
        }
        SyncResult.Failure -> {
          _feedbackEvents.send(UiFeedbackEvent(R.string.msg_sync_failed))
        }
      }
    }
  }

  private fun applySnapshot(snapshot: NotesSnapshot) {
    _uiState.update {
      it.copy(
        isLoading = false,
        allNotes = snapshot.allNotes,
        sqliteNotes = snapshot.sqliteNotes,
        sharedPrefsNotes = snapshot.sharedPrefsNotes,
        sqliteCount = snapshot.sqliteCount,
        sharedPrefsCount = snapshot.sharedPrefsCount,
        pendingSyncCount = snapshot.pendingSyncCount,
      )
    }
  }
}

@Composable
fun StorageBadge(
  displaySource: NoteDisplaySource,
  modifier: Modifier = Modifier,
) {
  val storageColors = LocalStorageColors.current
  val (labelRes, bgColor, fgColor, icon) =
    when (displaySource) {
      NoteDisplaySource.SQLITE_ONLY ->
        Quad(
          R.string.badge_sqlite,
          storageColors.sqliteContainer,
          storageColors.onSqliteContainer,
          Icons.Outlined.Storage,
        )
      NoteDisplaySource.SHARED_PREFS_ONLY ->
        Quad(
          R.string.badge_shared_prefs,
          storageColors.prefsContainer,
          storageColors.onPrefsContainer,
          Icons.Outlined.Settings,
        )
      NoteDisplaySource.SYNCED_IN_BOTH ->
        Quad(
          R.string.badge_synced,
          storageColors.syncedContainer,
          storageColors.onSyncedContainer,
          Icons.Outlined.CloudDone,
        )
    }

  Surface(
    modifier = modifier,
    shape = RoundedCornerShape(999.dp),
    color = bgColor,
  ) {
    Row(
      modifier = Modifier.padding(horizontal = 10.dp, vertical = 4.dp),
      verticalAlignment = Alignment.CenterVertically,
      horizontalArrangement = Arrangement.spacedBy(4.dp),
    ) {
      Icon(
        imageVector = icon,
        contentDescription = null,
        tint = fgColor,
        modifier = Modifier.size(14.dp),
      )
      Text(
        text = stringResource(labelRes),
        style = MaterialTheme.typography.labelMedium,
        color = fgColor,
      )
    }
  }
}

@Composable
fun StorageFilterTabs(
  selectedTab: NoteFilterTab,
  allCount: Int,
  sqliteCount: Int,
  sharedPrefsCount: Int,
  onSelectTab: (NoteFilterTab) -> Unit,
  modifier: Modifier = Modifier,
) {
  val tabs =
    listOf(
      Triple(NoteFilterTab.ALL, R.string.tab_all, allCount) to "tab_all",
      Triple(NoteFilterTab.SQLITE, R.string.tab_sqlite, sqliteCount) to "tab_sqlite",
      Triple(NoteFilterTab.SHARED_PREFERENCES, R.string.tab_shared_prefs, sharedPrefsCount) to
        "tab_shared_prefs",
    )

  Surface(
    modifier = modifier.fillMaxWidth(),
    shape = RoundedCornerShape(14.dp),
    color = MaterialTheme.colorScheme.surfaceVariant,
    border = BorderStroke(1.dp, MaterialTheme.colorScheme.outline),
  ) {
    Row(
      modifier = Modifier.fillMaxWidth().padding(4.dp),
      horizontalArrangement = Arrangement.spacedBy(4.dp),
    ) {
      tabs.forEach { (tabInfo, tag) ->
        val (tab, labelRes, count) = tabInfo
        val isSelected = selectedTab == tab
        val containerColor by
          animateColorAsState(
            targetValue =
              if (isSelected) MaterialTheme.colorScheme.surface else MaterialTheme.colorScheme.surfaceVariant,
            label = "tab_bg",
          )
        val textColor by
          animateColorAsState(
            targetValue =
              if (isSelected) MaterialTheme.colorScheme.primary
              else MaterialTheme.colorScheme.onSurfaceVariant,
            label = "tab_fg",
          )

        Box(
          modifier =
            Modifier.weight(1f)
              .minimumInteractiveComponentSize()
              .clip(RoundedCornerShape(10.dp))
              .background(containerColor)
              .clickable { onSelectTab(tab) }
              .testTag(tag)
              .padding(horizontal = 8.dp, vertical = 10.dp),
          contentAlignment = Alignment.Center,
        ) {
          Text(
            text = stringResource(R.string.tab_count_format, stringResource(labelRes), count),
            style = MaterialTheme.typography.labelLarge,
            color = textColor,
            maxLines = 1,
            overflow = TextOverflow.Ellipsis,
          )
        }
      }
    }
  }
}

@Composable
fun SyncStatusBanner(
  pendingSyncCount: Int,
  isSyncing: Boolean,
  modifier: Modifier = Modifier,
) {
  val storageColors = LocalStorageColors.current
  val hasPending = pendingSyncCount > 0
  val bgColor =
    if (hasPending) storageColors.pendingContainer else MaterialTheme.colorScheme.surfaceVariant
  val fgColor =
    if (hasPending) storageColors.onPendingContainer
    else MaterialTheme.colorScheme.onSurfaceVariant

  Surface(
    modifier = modifier.fillMaxWidth(),
    shape = RoundedCornerShape(12.dp),
    color = bgColor,
  ) {
    Row(
      modifier = Modifier.padding(horizontal = 12.dp, vertical = 8.dp),
      verticalAlignment = Alignment.CenterVertically,
      horizontalArrangement = Arrangement.spacedBy(8.dp),
    ) {
      Icon(
        imageVector = if (hasPending) Icons.Outlined.Sync else Icons.Outlined.CloudDone,
        contentDescription = null,
        tint = fgColor,
        modifier = Modifier.size(16.dp),
      )
      Text(
        text =
          when {
            isSyncing -> stringResource(R.string.sync_in_progress)
            hasPending -> stringResource(R.string.sync_status_pending_format, pendingSyncCount)
            else -> stringResource(R.string.sync_status_all_synced)
          },
        style = MaterialTheme.typography.labelMedium,
        color = fgColor,
      )
    }
  }
}

@Composable
fun NoteCard(
  note: Note,
  onClick: () -> Unit,
  onDeleteClick: () -> Unit,
  modifier: Modifier = Modifier,
) {
  val formattedDate = remember(note.updatedAt) { formatTimestamp(note.updatedAt) }
  val deleteCd = stringResource(R.string.cd_delete_note, note.title)

  Surface(
    modifier =
      modifier
        .fillMaxWidth()
        .clip(RoundedCornerShape(16.dp))
        .clickable(onClick = onClick)
        .testTag("note_card_${note.id}"),
    shape = RoundedCornerShape(16.dp),
    color = MaterialTheme.colorScheme.surface,
    tonalElevation = 1.dp,
    border = BorderStroke(1.dp, MaterialTheme.colorScheme.outline),
  ) {
    Column(modifier = Modifier.padding(start = 16.dp, top = 14.dp, end = 8.dp, bottom = 14.dp)) {
      Row(
        modifier = Modifier.fillMaxWidth(),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.SpaceBetween,
      ) {
        StorageBadge(displaySource = note.displaySource)
        IconButton(
          onClick = onDeleteClick,
          modifier = Modifier.minimumInteractiveComponentSize().testTag("delete_note_button_${note.id}"),
        ) {
          Icon(
            imageVector = Icons.Outlined.DeleteOutline,
            contentDescription = deleteCd,
            tint = MaterialTheme.colorScheme.onSurfaceVariant,
          )
        }
      }

      Spacer(modifier = Modifier.height(6.dp))

      Text(
        text = note.title,
        style = MaterialTheme.typography.titleMedium,
        color = MaterialTheme.colorScheme.onSurface,
        maxLines = 2,
        overflow = TextOverflow.Ellipsis,
        modifier = Modifier.padding(end = 8.dp),
      )

      Spacer(modifier = Modifier.height(6.dp))

      Text(
        text = note.content,
        style = MaterialTheme.typography.bodyMedium,
        color = MaterialTheme.colorScheme.onSurfaceVariant,
        maxLines = 3,
        overflow = TextOverflow.Ellipsis,
        modifier = Modifier.padding(end = 8.dp),

      Spacer(modifier = Modifier.height(12.dp))

      Text(
        text = stringResource(R.string.note_updated_format, formattedDate),
        style = MaterialTheme.typography.labelSmall,
        color = MaterialTheme.colorScheme.onSurfaceVariant,
      )
    }
  }
}

private fun formatTimestamp(epochMillis: Long): String {
  val formatter = SimpleDateFormat("MMM d, HH:mm", Locale.getDefault())
  return formatter.format(Date(epochMillis))
}

private fun StorageOptionCard(

import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.filled.Add
import androidx.compose.material.icons.outlined.Sync
import androidx.compose.material3.CircularProgressIndicator
import androidx.compose.material3.ExtendedFloatingActionButton
import androidx.compose.material3.FilledTonalButton
import androidx.compose.material3.Icon
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.Scaffold
import androidx.compose.material3.SnackbarHost
import androidx.compose.material3.SnackbarHostState
import androidx.compose.material3.Text
import androidx.compose.material3.minimumInteractiveComponentSize
import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.getValue
import androidx.compose.runtime.remember
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.platform.testTag
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import com.example.R

@Composable
fun NotesScreen(
  viewModel: NotesViewModel,
  modifier: Modifier = Modifier,
) {
  val uiState by viewModel.uiState.collectAsStateWithLifecycle()
  val snackbarHostState = remember { SnackbarHostState() }
  val context = LocalContext.current

  LaunchedEffect(viewModel, context) {
    viewModel.feedbackEvents.collect { event ->
      val message =
        if (event.formatArgs.isEmpty()) {
          context.getString(event.messageRes)
        } else {
          context.getString(event.messageRes, *event.formatArgs.toTypedArray())
        }
      snackbarHostState.showSnackbar(message)
    }
  }

  Scaffold(
    modifier = modifier.fillMaxSize(),
    containerColor = MaterialTheme.colorScheme.background,
    contentWindowInsets = WindowInsets.safeDrawing,
    snackbarHost = { SnackbarHost(hostState = snackbarHostState) },
    floatingActionButton = {
      ExtendedFloatingActionButton(
        onClick = viewModel::openCreateEditor,
        icon = {
          Icon(
            imageVector = Icons.Filled.Add,
            contentDescription = stringResource(R.string.cd_add_note),
          )
        },
        text = { Text(text = stringResource(R.string.action_new_note)) },
        containerColor = MaterialTheme.colorScheme.primary,
        contentColor = MaterialTheme.colorScheme.onPrimary,
        modifier = Modifier.testTag("fab_add_note"),
      )
    },
  ) { innerPadding ->
    Box(
      modifier = Modifier.fillMaxSize().padding(innerPadding),
      contentAlignment = Alignment.TopCenter,
    ) {
      Column(
        modifier = Modifier.widthIn(max = 640.dp).fillMaxSize().padding(horizontal = 16.dp),
      ) {
        Spacer(modifier = Modifier.height(12.dp))

        Row(
          modifier = Modifier.fillMaxWidth(),
          verticalAlignment = Alignment.CenterVertically,
          horizontalArrangement = Arrangement.SpaceBetween,
        ) {
          Column(modifier = Modifier.weight(1f)) {
            Text(
              text = stringResource(R.string.app_name),
              style = MaterialTheme.typography.headlineMedium,
              color = MaterialTheme.colorScheme.onBackground,
            )
            Text(
              text = stringResource(R.string.app_subtitle),
              style = MaterialTheme.typography.bodyMedium,
              color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
          }

          Spacer(modifier = Modifier.width(12.dp))

          FilledTonalButton(
            onClick = viewModel::syncNotes,
            enabled = !uiState.isSyncing,
            shape = RoundedCornerShape(12.dp),
            modifier = Modifier.minimumInteractiveComponentSize().testTag("sync_button"),
          ) {
            if (uiState.isSyncing) {
              CircularProgressIndicator(
                modifier = Modifier.size(16.dp),
                strokeWidth = 2.dp,
              )
            } else {
              Icon(
                imageVector = Icons.Outlined.Sync,
                contentDescription = stringResource(R.string.cd_sync_notes),
                modifier = Modifier.size(18.dp),
              )
            }
            Spacer(modifier = Modifier.width(6.dp))
            Text(text = stringResource(R.string.action_sync))
          }
        }

        Spacer(modifier = Modifier.height(14.dp))

        SyncStatusBanner(
          pendingSyncCount = uiState.pendingSyncCount,
          isSyncing = uiState.isSyncing,
        )

        Spacer(modifier = Modifier.height(14.dp))

        StorageFilterTabs(
          selectedTab = uiState.selectedTab,
          allCount = uiState.allCount,
          sqliteCount = uiState.sqliteCount,
          sharedPrefsCount = uiState.sharedPrefsCount,
          onSelectTab = viewModel::selectTab,
        )

        Spacer(modifier = Modifier.height(12.dp))

        AnimatedVisibility(
          visible = !uiState.isLoading && uiState.displayedNotes.isEmpty(),
          enter = fadeIn(),
          exit = fadeOut(),
        ) {
          EmptyNotesView(selectedTab = uiState.selectedTab)
        }

        AnimatedVisibility(
          visible = uiState.displayedNotes.isNotEmpty(),
          enter = fadeIn(),
          exit = fadeOut(),
          modifier = Modifier.weight(1f),
        ) {
          LazyColumn(
            modifier = Modifier.fillMaxSize().testTag("notes_list"),
            contentPadding = PaddingValues(top = 4.dp, bottom = 96.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp),
          ) {
            items(
              items = uiState.displayedNotes,
              key = { note -> "${note.storageMethod.name}_${note.id}" },
            ) { note ->
              NoteCard(
                note = note,
                onClick = { viewModel.openEditEditor(note) },
                onDeleteClick = { viewModel.requestDeleteNote(note) },
              )
            }
          }
        }
      }
    }
  }

  if (uiState.editorState.isVisible) {
    NoteEditorSheet(
      editorState = uiState.editorState,
      onTitleChange = viewModel::onEditorTitleChanged,
      onContentChange = viewModel::onEditorContentChanged,
      onStorageSelect = viewModel::onEditorStorageSelected,
      onSave = viewModel::saveNote,
      onDismiss = viewModel::dismissEditor,
    )
  }

  uiState.notePendingDelete?.let { pendingNote ->
    DeleteNoteDialog(
      note = pendingNote,
      onConfirm = viewModel::confirmDeleteNote,
      onDismiss = viewModel::dismissDeleteDialog,
    )
  }
}
